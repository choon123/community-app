from flask import Flask, render_template, request, redirect, session import os import psycopg2 app = Flask(__name__) app.secret_key = "secret123" # 🔗 Database Connection (SAFE VERSION) DATABASE_URL = os.environ.get("DATABASE_URL") def get_conn(): return psycopg2.connect(DATABASE_URL) # 🛠️ Create Tables conn = get_conn() cursor = conn.cursor() cursor.execute(""" CREATE TABLE IF NOT EXISTS users ( id SERIAL PRIMARY KEY, username TEXT, password TEXT ); """) cursor.execute(""" CREATE TABLE IF NOT EXISTS jobs ( id SERIAL PRIMARY KEY, title TEXT, location TEXT, contact TEXT, salary TEXT, workers TEXT, meals TEXT, description TEXT ); """) # ⭐ NEW TABLE (SAVE JOBS) cursor.execute(""" CREATE TABLE IF NOT EXISTS saved_jobs ( id SERIAL PRIMARY KEY, username TEXT, job_id INTEGER ); """) conn.commit() conn.close() # 🌐 LANGUAGE SYSTEM translations = { "en": { "title": "Community Help App", "register": "Register", "login": "Login", "jobs": "Jobs", "post": "Post Job", "logout": "Logout", "contacts": "Emergency", "saved": "Saved Jobs", "welcome_login": "Welcome back! Keep going 💪", "welcome_register": "Welcome! Your journey starts 🚀" } } def get_lang(): return session.get('lang', 'en') @app.route('/set_language/') def set_language(lang): session['lang'] = lang return redirect('/') # 🏠 Home @app.route('/') def index(): t = translations["en"] message = session.pop('message', None) return render_template('index.html', t=t, message=message) # 📋 Jobs (with SEARCH) @app.route('/jobs') def jobs(): search = request.args.get('search') location = request.args.get('location') conn = get_conn() cursor = conn.cursor() query = "SELECT * FROM jobs WHERE 1=1" values = [] if search: query += " AND title ILIKE %s" values.append(f"%{search}%") if location: query += " AND location ILIKE %s" values.append(f"%{location}%") query += " ORDER BY id DESC" cursor.execute(query, values) data = cursor.fetchall() conn.close() return render_template('jobs.html', jobs=data) # ➕ Post Job @app.route('/post', methods=['GET', 'POST']) def post_job(): if 'user' not in session: return redirect('/login') if request.method == 'POST': conn = get_conn() cursor = conn.cursor() cursor.execute(""" INSERT INTO jobs (title, location, contact, salary, workers, meals, description) VALUES (%s, %s, %s, %s, %s, %s, %s) """, ( request.form['title'], request.form['location'], request.form['contact'], request.form['salary'], request.form['workers'], request.form['meals'], request.form['description'] )) conn.commit() conn.close() return redirect('/jobs') return render_template('post_job.html') # ❤️ SAVE JOB @app.route('/save/') def save_job(job_id): if 'user' not in session: return redirect('/login') conn = get_conn() cursor = conn.cursor() cursor.execute(""" INSERT INTO saved_jobs (username, job_id) VALUES (%s, %s) """, (session['user'], job_id)) conn.commit() conn.close() return redirect('/jobs') # ❤️ VIEW SAVED JOBS @app.route('/saved') def saved_jobs(): if 'user' not in session: return redirect('/login') conn = get_conn() cursor = conn.cursor() cursor.execute(""" SELECT jobs.* FROM jobs JOIN saved_jobs ON jobs.id = saved_jobs.job_id WHERE saved_jobs.username = %s ORDER BY jobs.id DESC """, (session['user'],)) data = cursor.fetchall() conn.close() return render_template('jobs.html', jobs=data) # 📝 Register @app.route('/register', methods=['GET', 'POST']) def register(): if request.method == 'POST': conn = get_conn() cursor = conn.cursor() cursor.execute(""" INSERT INTO users (username, password) VALUES (%s, %s) """, (request.form['username'], request.form['password'])) conn.commit() conn.close() session['message'] = "Welcome! 🚀" return redirect('/') return render_template('register.html') # 🔐 Login @app.route('/login', methods=['GET', 'POST']) def login(): if request.method == 'POST': conn = get_conn() cursor = conn.cursor() cursor.execute(""" SELECT * FROM users WHERE username=%s AND password=%s """, (request.form['username'], request.form['password'])) user = cursor.fetchone() conn.close() if user: session['user'] = request.form['username'] session['message'] = "Welcome back 💪" return redirect('/') else: return "Invalid login" return render_template('login.html') # 🚪 Logout @app.route('/logout') def logout(): session.pop('user', None) return redirect('/') # 📞 Contacts @app.route('/contacts') def contacts(): return render_template('contacts.html') if __name__ == '__main__': app.run(debug=True)
